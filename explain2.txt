    ], default='basic')

    access_start_time = models.TimeField(default='09:00:00')
    access_end_time = models.TimeField(default='17:00:00')

    def can_access_now(self):
        from django.utils import timezone
        current_time = timezone.now().time()
        return self.access_start_time <= current_time <= self.access_end_time

# 2. Custom permissions
class AnalystPermission(BasePermission):
    def has_permission(self, request, view):
        if not hasattr(request.user, 'analyst'):
            return False
        
        analyst = request.user.analyst
        if not analyst.can_access_now:
            return False
            
        # Check report-specific permissions
        if hasattr(view, 'required_report_type'):
            return analyst.allowed_reports.filter(
                name=view.required_report_type
            ).exists()
        
        return True

class Permission(models.Model):
    name = models.CharField(max_length=100, unique=True)
    description = models.TextField()

    action = models.CharField(max_length=20)    # 'create', 'read', 'update', 'delete'

class RolePermission(models.Model):
    role = models.CharField(max_length=20, choices=ROLE_CHOICES)
    permission = models.ForeignKey(Permission, on_delete=models.CASCADE)
    granted = models.BooleanField(default=True)  # Can explicitly deny permissions

# Custom permission checking
def user_can_perform_action(user, resource, action):
    role = user.profile.role
    
    permission_granted = RolePermission.objects.filter(
        role=role,
        permission__resource=resource,
        permission__action=action,
        granted=True
    ).exists()
    
    permission_denied = RolePermission.objects.filter(
        role=role,
        permission__resource=resource,
        permission__action=action,
        granted=False
    ).exists()
    
    return permission_granted and not permission_denied
